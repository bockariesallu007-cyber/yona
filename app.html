async function saveMember(event) { event.preventDefault(); const button = document.getElementById("memberSaveButton"); button.disabled = true; button.textContent = editingMemberId ? "Updating..." : "Saving..."; try { /* ===================================================== CHECK LOGIN SESSION ===================================================== */ const { data: sessionData, error: sessionError } = await db.auth.getSession(); if (sessionError) { throw new Error( "Could not check your login session: " + sessionError.message ); } if ( !sessionData || !sessionData.session ) { throw new Error( "Your login session has expired. Please log in again." ); } /* ===================================================== GET MEMBER INFORMATION ===================================================== */ const name = document .getElementById("memberName") .value .trim(); const phone = document .getElementById("memberPhone") .value .trim(); const memberType = document .getElementById("memberType") .value; const position = document .getElementById("memberPosition") .value .trim(); const photoInput = document.getElementById("memberPhoto"); const file = photoInput.files && photoInput.files[0]; /* ===================================================== VALIDATION ===================================================== */ if (!name) { throw new Error( "Member name is required." ); } if (!memberType) { throw new Error( "Please select the member category." ); } if (file && !validatePhoto(file)) { return; } /* ===================================================== PHOTO ===================================================== */ let newPhotoPath = editingMemberOldPhoto || null; if (file) { newPhotoPath = await uploadPersonPhoto( file, "member-photos", "member" ); } /* ===================================================== MEMBER RECORD ===================================================== */ const record = { name: name, phone: phone, position: position, member_type: memberType, photo_url: newPhotoPath }; console.log( "Saving YONA member:", record ); /* ===================================================== UPDATE EXISTING MEMBER ===================================================== */ if (editingMemberId) { const { error } = await db .from("members") .update(record) .eq( "id", editingMemberId ); if (error) { console.error( "MEMBER UPDATE ERROR:", error ); if ( file && newPhotoPath && newPhotoPath !== editingMemberOldPhoto ) { await deleteStoredPhoto( newPhotoPath ); } throw new Error( `Could not update member. Code: ${error.code || "N/A"} Message: ${error.message || "Unknown error"} Details: ${error.details || "None"} Hint: ${error.hint || "None"}` ); } /* Delete old photo only after successful update */ if ( file && editingMemberOldPhoto && editingMemberOldPhoto !== newPhotoPath ) { await deleteStoredPhoto( editingMemberOldPhoto ); } showMessage( "Member updated successfully." ); } /* ===================================================== INSERT NEW MEMBER ===================================================== */ else { const { error } = await db .from("members") .insert([record]); if (error) { console.error( "MEMBER INSERT ERROR:", error ); if (newPhotoPath) { await deleteStoredPhoto( newPhotoPath ); } throw new Error( `Could not save member. Code: ${error.code || "N/A"} Message: ${error.message || "Unknown error"} Details: ${error.details || "None"} Hint: ${error.hint || "None"}` ); } console.log( "YONA member saved successfully." ); showMessage( "Member saved successfully." ); } /* ===================================================== RESET FORM ===================================================== */ resetMemberForm(); /* ===================================================== RELOAD MEMBERS ===================================================== */ await loadMembers(); } catch (error) { console.error( "YONA MEMBER SAVE FAILED:", error ); let message = error && error.message ? error.message : "Unknown error occurred."; /* Special message for browser/network failure */ if ( message === "Failed to fetch" || message.toLowerCase().includes("failed to fetch") ) { message = "The app could not connect to Supabase.\n\n" + "Please check your internet connection and make sure the Supabase project is online."; } showMessage( "Could not save member:\n" + message, "error" ); } finally { button.disabled = false; button.textContent = editingMemberId ? "Update Member" : "Save Member"; } }
